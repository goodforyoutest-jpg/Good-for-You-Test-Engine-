async function loadLiveTests() { const container = document.getElementById('liveTestsContainer'); try { const userOrgCode = userDoc?.orgCode || null; // Include orgCode in the query because firestore.rules requires every // returned test to belong to the signed-in user's organisation. const snap = await window.db.collection('tests') .where('active','==',true) .where('orgCode','==',userOrgCode) .get(); const orgDocs = snap.docs; if (orgDocs.length === 0) { container.innerHTML = `

No live tests available right now.
Check back soon or take your own test.

`; return; } const resultsSnap = await window.db.collection('results') .where('userId','==',currentUser.uid).get(); const attemptedMap = {}; resultsSnap.docs.forEach(d => { const r = d.data(); if (r.testId && !attemptedMap[r.testId]) { attemptedMap[r.testId] = { resultId: d.id, pct: r.percentage || 0 }; } }); container.innerHTML = orgDocs.map(doc => { const t = doc.data(); const mins = t.durationMinutes || '—'; const attempted = attemptedMap[doc.id]; if (attempted) { return `
LIVE
${escHtml(t.title)}
${(t.questions||[]).length} Questions · ${mins} mins · ${t.subject||'General'}
VIEW RESULT →
Score: ${attempted.pct}%
`; } return `
LIVE
${escHtml(t.title)}
${(t.questions||[]).length} Questions · ${mins} mins · ${t.subject||'General'}
JOIN →
`; }).join(''); } catch(e) { container.innerHTML = `

Error loading tests. Check Firebase config.

`; console.error('loadLiveTests failed:', e); } }
